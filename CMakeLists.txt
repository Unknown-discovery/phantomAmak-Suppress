cmake_minimum_required(VERSION 3.16)
project(amak VERSION 0.1.0 LANGUAGES CXX)

option(AMAK_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(AMAK_ENABLE_AVX2 "Build optional runtime-dispatched AVX2 DCT kernels on supported toolchains" ON)
find_package(Threads REQUIRED)

add_library(amak_runtime src/dct.cpp src/dct_kernels.cpp src/rans.cpp src/container.cpp src/runtime.cpp)
target_include_directories(amak_runtime PUBLIC include PRIVATE src)
target_compile_features(amak_runtime PUBLIC cxx_std_17)
target_compile_definitions(amak_runtime PRIVATE AMAK_ENABLE_AVX2=$<BOOL:${AMAK_ENABLE_AVX2}>)
target_link_libraries(amak_runtime PUBLIC Threads::Threads)

add_executable(amak src/main.cpp src/benchmark.cpp)
target_include_directories(amak PRIVATE src)
target_link_libraries(amak PRIVATE amak_runtime)

include(CTest)
if(BUILD_TESTING)
    add_executable(amak_tests tests/tests.cpp)
    target_include_directories(amak_tests PRIVATE src)
    target_link_libraries(amak_tests PRIVATE amak_runtime)
    add_test(NAME unit COMMAND amak_tests)
    set_tests_properties(unit PROPERTIES TIMEOUT 120)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli_and_reference COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_smoke.py $<TARGET_FILE:amak>)
    set_tests_properties(cli_and_reference PROPERTIES TIMEOUT 120)
endif()

foreach(target amak_runtime amak amak_tests)
    if(TARGET ${target})
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -ffp-contract=off)
            if(AMAK_SANITIZE)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
    endif()
endforeach()
